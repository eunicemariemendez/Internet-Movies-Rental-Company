-- Create movies table
create table if not exists public.movies (
  id bigint generated by default as identity primary key,
  title text not null,
  actors text[] not null,
  release_year integer not null
);

-- Enable Row Level Security
alter table public.movies enable row level security;

-- Create policies
create policy "Allow read access to movies"
  on public.movies
  for select
  using (true);

create policy "Allow insert access to movies"
  on public.movies
  for insert
  with check (true);

create policy "Allow update access to movies"
  on public.movies
  for update
  using (true)
  with check (true);

create policy "Allow delete access to movies"
  on public.movies
  for delete
  using (true);

-- Insert movie titles
insert into public.movies (title, actors, release_year)
select v.title, v.actors, v.release_year
from (
  values
    ('Inception', array['Leonardo DiCaprio', 'Elliot Page', 'Joseph Gordon-Levitt'], 2010),
    ('The Martian', array['Matt Damon', 'Jessica Chastain', 'Donald Glover','Michael Peña'], 2015),
    ('The Matrix', array['Keanu Reeves', 'Laurence Fishburne', 'Carrie-Anne Moss'], 1999),
    ('Interstellar', array['Matthew McConaughey', 'Anne Hathaway', 'Jessica Chastain'], 2014),
    ('Spider-Man: Brand New Day', array['Tom Holland', 'Sadie Sink', 'Zendaya'], 2026)
) as v(title, actors, release_year)
where not exists (select 1 from public.movies limit 1);
